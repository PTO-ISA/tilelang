# from tvm.script import tirx as T

@T.prim_func
def main(X_handle: T.handle, Out_handle: T.handle):
    X = T.match_buffer(X_handle, (64, 128), "float16", strides=(128, 1))
    Out = T.match_buffer(Out_handle, (64, 128), "float16", strides=(128, 1))
    # with T.sblock("root"):
    for bx in T.thread_binding(1, thread="blockIdx.x"):
        for tx in T.thread_binding(128, thread="threadIdx.x"):
            for ty in T.thread_binding(1, thread="threadIdx.y"):
                for tz in T.thread_binding(1, thread="threadIdx.z"):
                    with T.sblock("tilelang_root"):
                        T.reads()
                        T.writes()
                        x_ub = T.sblock_alloc_buffer((64, 128), "float16", scope="shared.dyn")
                        out_ub = T.sblock_alloc_buffer((64, 128), "float16", scope="shared.dyn")
                        T.copy(T.region(X[0, 0], 1, 64, 128), T.region(x_ub[0, 0], 2, 64, 128))
                        with T.sblock("SIMT_VF", no_realize=True):
                            T.reads()
                            T.writes()
                            T.sblock_attr({"tl.vf_source_index": T.int64(0)})
                            sf_inv = T.sblock_alloc_buffer((64, 8), scope="local.fragment")
                            m = T.sblock_alloc_buffer((1,), scope="local.var")
                            s = T.sblock_alloc_buffer((1,), scope="local.var")
                            simtvf_tx = T.launch_thread("threadIdx.x", 32)
                            simtvf_ty = T.launch_thread("threadIdx.y", 1)
                            simtvf_tz = T.launch_thread("threadIdx.z", 1)
                            T.attr("simtvf", "tl.simtvf_scope", 1)
                            for i in T.parallel(64):
                                for g in T.parallel(8):
                                    m[0] = T.float32(0.0)
                                    for t in range(16):
                                        vv: T.float32 = T.Cast("float32", x_ub[i, g * 16 + t])
                                        m[0] = T.max(m[0], T.max(vv, vv * T.float32(-1.0)))
                                    m[0] = T.max(m[0], T.float32(9.9999999999999995e-07))
                                    sf_inv[i, g] = T.float32(1.0) / m[0]
                            for i in T.parallel(64):
                                for g in T.parallel(8):
                                    s[0] = sf_inv[i, g]
                                    for t in range(16):
                                        j: T.int32 = g * 16 + t
                                        out_ub[i, j] = T.Cast("float16", T.Cast("float32", x_ub[i, j]) * s[0])
                        T.copy(T.region(out_ub[0, 0], 1, 64, 128), T.region(Out[0, 0], 2, 64, 128))
