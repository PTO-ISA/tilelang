# from tvm.script import tirx as T

@T.prim_func
def main(A_handle: T.handle, X_handle: T.handle, Out_handle: T.handle):
    A = T.match_buffer(A_handle, (16, 32, 64), "float16", strides=(2048, 64, 1))
    X = T.match_buffer(X_handle, (16,), "float16", strides=(1,))
    Out = T.match_buffer(Out_handle, (32, 64), strides=(64, 1))
    # with T.sblock("root"):
    for bx in T.thread_binding(1, thread="blockIdx.x"):
        for tx in T.thread_binding(128, thread="threadIdx.x"):
            for ty in T.thread_binding(1, thread="threadIdx.y"):
                for tz in T.thread_binding(1, thread="threadIdx.z"):
                    with T.sblock("tilelang_root"):
                        T.reads()
                        T.writes()
                        a_ub = T.sblock_alloc_buffer((16, 32, 64), "float16", scope="shared.dyn")
                        x_ub = T.sblock_alloc_buffer((16,), "float16", scope="shared.dyn")
                        out_ub = T.sblock_alloc_buffer((32, 64), scope="shared.dyn")
                        T.copy(T.region(A[0, 0, 0], 1, 16, 32, 64), T.region(a_ub[0, 0, 0], 2, 16, 32, 64))
                        T.copy(T.region(X[0], 1, 16), T.region(x_ub[0], 2, 16))
                        with T.sblock("SIMT_VF", no_realize=True):
                            T.reads()
                            T.writes()
                            T.sblock_attr({"tl.vf_source_index": T.int64(0)})
                            acc_c = T.sblock_alloc_buffer((16, 64), scope="local.fragment")
                            xk = T.sblock_alloc_buffer((1,), scope="local.var")
                            simtvf_tx = T.launch_thread("threadIdx.x", 32)
                            simtvf_ty = T.launch_thread("threadIdx.y", 1)
                            simtvf_tz = T.launch_thread("threadIdx.z", 1)
                            T.attr("simtvf", "tl.simtvf_scope", 1)
                            for c in range(2):
                                for i in T.parallel(16):
                                    for j in T.parallel(64):
                                        acc_c[i, j] = T.float32(0.0)
                                for k in range(16):
                                    xk[0] = T.Cast("float32", x_ub[k])
                                    for i in T.parallel(16):
                                        for j in T.parallel(64):
                                            row: T.int32 = c * 16 + i
                                            if row < 32:
                                                acc_c[i, j] = acc_c[i, j] + T.Cast("float32", a_ub[k, row, j]) * xk[0]
                                for i in T.parallel(16):
                                    for j in T.parallel(64):
                                        row: T.int32 = c * 16 + i
                                        if row < 32:
                                            out_ub[row, j] = acc_c[i, j]
                        T.copy(T.region(out_ub[0, 0], 1, 32, 64), T.region(Out[0, 0], 2, 32, 64))
